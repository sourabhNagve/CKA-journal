# deployment yaml
apiVersion: apps/v1
kind: Deployment
metadata:
  name: mysql
  namespace: mysql
  labels:
    app: mysql
    tier: database
spec:
  replicas: 1
  selector:
    matchLabels:
      app: mysql
  template:
    metadata:
      labels:
        app: mysql
        tier: database
    spec:
      volumes:
      - name: shared
        persistentVolumeClaim:
         claimName: mysql-pvc
      containers:
        - name: mysql
          image: mysql:5.7
          env:
            - name: MYSQL_ROOT_PASSWORD
              value: "rootpassword123"
            - name: MYSQL_DATABASE
              value: "customerdb"
          ports:
            - containerPort: 3306
              name: mysql
          volumeMounts:
          - name: shared
            mountPath: /home/data

-----------------
# pvc.yaml
apiVersion: v1
kind: PersistentVolumeClaim
metadata:
  name: mysql-pvc
  namespace: mysql
spec:
  accessModes:
    - ReadWriteOnce
  volumeMode: Filesystem
  resources:
    requests:
      storage: 250Mi
  storageClassName: ""

# pv yaml
apiVersion: v1
  kind: PersistentVolume
  metadata:
    finalizers:
    - kubernetes.io/pv-protection
    labels:
      type: local
    name: mysql-pv-retain
  spec:
    accessModes:
    - ReadWriteOnce
    capacity:
      storage: 500Mi
    hostPath:
      path: /mnt/mysql-data
      type: DirectoryOrCreate
    persistentVolumeReclaimPolicy: Retain
    volumeMode: Filesystem

----------------------------------

- we removed the claimRef section in the pv edit
- applied the pvc, so the pvc is bound to the pv
- edited the deployment yaml and added the volume which is the pvc and volume mounts.
